function editCustomer(id){
  const x=allCustomers.find(c=>c.id===id);if(!x)return showMsg('Müşteri bulunamadı.','err');
  $('c_id').value=x.id;$('customerModalTitle').textContent='Müşteri Güncelle';$('c_company').value=x.company||'';$('c_name').value=x.contact_name||'';
  let ph=[],em=[];try{ph=JSON.parse(x.phones_json||'[]')}catch{}try{em=JSON.parse(x.emails_json||'[]')}catch{}
  if(!ph.length&&x.phone)ph=[x.phone];if(!em.length&&x.email)em=[x.email];
  $('c_phones').value=ph.join('\n');$('c_emails').value=em.join('\n');$('c_region').value=x.region||'';$('c_priority').value=x.priority||'NORMAL';$('c_stage').value=x.stage||'Yeni Lead';$('c_follow').value=x.follow_date||'';
  $('c_special_notes').value=x.special_notes||'';$('c_machine_info').value=x.machine_info||'';$('c_invoice_title').value=x.invoice_title||'';$('c_tax_office').value=x.tax_office||'';$('c_tax_number').value=x.tax_number||'';$('c_invoice_address').value=x.invoice_address||'';
  const cats=(x.categories||x.sector||'').split(',').map(s=>s.trim()).filter(Boolean);document.querySelectorAll('.cat-check').forEach(c=>c.checked=cats.includes(c.value));

}

async function saveCustomer(){
  try{
    const company=$('c_company').value.trim();if(!company)return showMsg('Firma adı zorunlu.','err');
    const phones=$('c_phones').value.split(/\n|,/).map(s=>s.trim()).filter(Boolean);
    const emails=$('c_emails').value.split(/\n|,/).map(s=>s.trim()).filter(Boolean);
    const categories=[...document.querySelectorAll('.cat-check:checked')].map(c=>c.value);
    const payload={company,contact_name:$('c_name').value.trim(),phones,emails,region:$('c_region').value.trim(),priority:$('c_priority').value,stage:$('c_stage').value,follow_date:$('c_follow').value,categories,special_notes:$('c_special_notes').value.trim(),machine_info:$('c_machine_info').value.trim(),invoice_title:$('c_invoice_title').value.trim(),tax_office:$('c_tax_office').value.trim(),tax_number:$('c_tax_number').value.trim(),invoice_address:$('c_invoice_address').value.trim()};
    const id=Number($('c_id').value||0);
    if(id)await req('/api/customers/'+id,{method:'PUT',body:JSON.stringify(payload)});else await req('/api/customers',{method:'POST',body:JSON.stringify(payload)});
    closeM('customerModal');showMsg(id?'Müşteri güncellendi.':'Müşteri kaydedildi.');await loadAll();
  }catch(e){showMsg(e.message,'err')}
}
